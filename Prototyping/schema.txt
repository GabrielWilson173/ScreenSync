CREATE TABLE Users (
                username TEXT UNIQUE,
                password_hash TEXT,
                uuid CHAR(36) PRIMARY KEY
            );
CREATE TABLE User_data (
                uuid CHAR(36),
                data TEXT,
                FOREIGN KEY (uuid) REFERENCES Users (uuid)
            );
CREATE TABLE Screentime(
    uuid CHAR(36),
    hwid CHAR(36),
    app_name TEXT,
    seconds_spent FLOAT,
    last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (uuid, hwid, app_name)
);
CREATE TABLE devices(
    uuid CHAR(36),
    hwid CHAR(36),
    device_num INT,
    PRIMARY KEY (uuid, hwid)
);
